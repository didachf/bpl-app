# El relieve en el motor, plan de implementación

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** que el motor sepa dónde está el suelo: mínimo de 100 ft sobre el terreno bajo el recorrido, terra a ras siguiendo el suelo, y la zona y los planes cortados donde dejan de valer, como manda `docs/superpowers/specs/2026-11-01-relieve-design.md`.

**Architecture:** un módulo nuevo, `relieve.ts`, consulta una rejilla de elevación en el plano local del motor (bilineal para el suelo, máximo de la celda para comprobar el mínimo). Las tablas de `vertical.ts` guardan el terreno, la altitud de cada minuto de cada cambio y `vale()`, la comprobación única. `plan.ts` corta cada plan en su primera posición no válida con la misma regla en la búsqueda, el refinado y `limite()`. `zona.ts` hace dos pasadas: la cota exterior sin relieve y la zona pintada podando. `motor.ts` pone terra al suelo y las franjas por encima.

**Tech Stack:** TypeScript (`tsc -b`), vitest (`npm test`, con `TZ=Europe/Madrid`), sin dependencias nuevas.

**Convenciones:** las del plan del objetivo de altura. Rama `terra-al-punt`, encima de `main`, con 584 pruebas en verde.

**La regla de validez, una sola vez** (spec §4.3), que todas las piezas aplican igual:
* Al final de cada minuto `k ≥ 1`, nunca en el minuto 0 (donde está el globo es un hecho).
* Manteniendo una franja que no es terra: su altitud.
* En un cambio entre dos franjas que no son terra: la altitud de ese minuto. En un cambio que sale de
  terra o va a terra: sólo el último minuto, y sólo si la franja de llegada no es terra.
* El plan cuenta hasta su primera posición no válida `L`: posiciones `0..L-1`, segmentos `0..L-2`.

```ts
// src/services/nav/relieve.ts
// El suelo bajo el globo (spec 2026-11-01-relieve §4.1): una rejilla de elevacion en grados,
// consultada en grados o en el plano local del motor. Bilineal para el suelo, y el maximo de la
// celda para comprobar el minimo sobre el terreno, que deja el error del lado seguro.
import type { Plano } from './geo'
import type { Relieve } from './tipos'

const R = 6371000
const RAD = Math.PI / 180
const EPS = 1e-9

/** El suelo en el plano local, m sobre el mar. NaN fuera de la rejilla. */
export type Suelo = (x: number, y: number) => number

/** Lo que las tablas necesitan del relieve: el techo de cada celda y el suelo mas alto. */
export interface Terreno { techo: Suelo; maxM: number }

function celda(r: Relieve, fi: number, fj: number, max: boolean): number {
  if (!(fi >= -EPS && fj >= -EPS && fi <= r.nLat - 1 + EPS && fj <= r.nLon - 1 + EPS)) return NaN
  const i = Math.min(Math.max(Math.floor(fi), 0), r.nLat - 2)
  const j = Math.min(Math.max(Math.floor(fj), 0), r.nLon - 2)
  const n = r.nLon
  const a = r.elevM[i * n + j]
  const b = r.elevM[i * n + j + 1]
  const c = r.elevM[(i + 1) * n + j]
  const d = r.elevM[(i + 1) * n + j + 1]
  if (max) return Math.max(a, b, c, d)
  const u = Math.min(Math.max(fj - j, 0), 1)
  const v = Math.min(Math.max(fi - i, 0), 1)
  return a * (1 - u) * (1 - v) + b * u * (1 - v) + c * (1 - u) * v + d * u * v
}

export function sueloGeo(r: Relieve, lat: number, lon: number): number {
  return celda(r, (lat - r.lat0) / r.dLat, (lon - r.lon0) / r.dLon, false)
}

// aGeo es lineal en el plano: lat = lat0 + y / (R RAD), lon = lon0 + x / (R RAD cos lat0).
function enPlano(r: Relieve, p: Plano, max: boolean): Suelo {
  const i0 = (p.lat0 - r.lat0) / r.dLat
  const iy = 1 / (R * RAD * r.dLat)
  const j0 = (p.lon0 - r.lon0) / r.dLon
  const jx = 1 / (R * RAD * Math.cos(p.lat0 * RAD) * r.dLon)
  return (x, y) => celda(r, i0 + y * iy, j0 + x * jx, max)
}

export function sueloPlano(r: Relieve, p: Plano): Suelo {
  return enPlano(r, p, false)
}

/** El maximo de los cuatro nodos de la celda: el que se usa para el minimo sobre el terreno. */
export function techoPlano(r: Relieve, p: Plano): Suelo {
  return enPlano(r, p, true)
}

export function maximo(r: Relieve): number {
  let m = -Infinity
  for (const z of r.elevM) if (z > m) m = z
  return m
}
```

```ts
/** Un relieve en el plano de `p`, con nodos cada `paso` m de (x0, y0) a (x1, y1) y el suelo de `f`. */
export function relieveSintetico(
  p: Plano, x0: number, y0: number, x1: number, y1: number, paso: number, f: (x: number, y: number) => number,
): Relieve {
  const R = 6371000
  const RAD = Math.PI / 180
  const nLat = Math.round((y1 - y0) / paso) + 1
  const nLon = Math.round((x1 - x0) / paso) + 1
  const o = aGeo(p, x0, y0)
  const elevM: number[] = []
  for (let i = 0; i < nLat; i++) for (let j = 0; j < nLon; j++) elevM.push(f(x0 + j * paso, y0 + i * paso))
  return { lat0: o.lat, lon0: o.lon, dLat: paso / (R * RAD), dLon: paso / (R * RAD * Math.cos(p.lat0 * RAD)), nLat, nLon, elevM }
}
```

```ts
export function zTr(t: Tablas, b: number, b2: number, cl: number): number[] {
  return t.altTr[(b * t.c.rejilla.altM.length + b2) * 2 + cl]
}

/**
 * La altitud que hay que comprobar al final del minuto j del cambio de b a b2, o NaN si ninguna. Entre
 * franjas, la de cada minuto. Saliendo de terra o yendo a ella, solo la llegada a una franja que no es
 * terra (spec 2026-11-01-relieve §4.3).
 */
export function zComprobar(t: Tablas, b: number, b2: number, cl: number, j: number): number {
  const zt = zTr(t, b, b2, cl)
  if (b !== t.iTerra && b2 !== t.iTerra) return zt[j]
  return j === zt.length - 1 && b2 !== t.iTerra ? zt[j] : NaN
}

/** Si la altitud z en (x, y) queda al menos el minimo sobre el techo del suelo. Sin suelo conocido, vale. */
export function vale(t: Tablas, z: number, x: number, y: number): boolean {
  if (z >= t.alturaLibre) return true
  return !(z < t.techo!(x, y) + t.minM - 1e-6)
}
```

```ts
/**
 * El primer minuto en que el plan deja de valer sobre el terreno, H + 1 si vale entero
 * (spec 2026-11-01-relieve §4.3). `cam` es su camino, con las mismas tablas.
 */
export function limite(t: Tablas, p: PlanInterno, cam: Camino): number {
  const H = t.c.minutos
  const z = t.c.rejilla.altM
  let k = 0
  for (let i = 0; i < p.franjas.length; i++) {
    const b = p.franjas[i]
    if (i > 0) {
      const a = p.franjas[i - 1]
      const n = fr(t, a, b, p.clases[i - 1]).length
      for (let j = 0; j < n; j++) {
        if (++k > H) return H + 1
        const zc = zComprobar(t, a, b, p.clases[i - 1], j)
        if (zc === zc && !vale(t, zc, cam[k][0], cam[k][1])) return k
      }
    }
    const fin = i === p.franjas.length - 1 ? H : k + p.dur[i]
    while (k < fin) {
      k++
      if (b !== t.iTerra && !vale(t, z[b], cam[k][0], cam[k][1])) return k
    }
  }
  return H + 1
}

/** La distancia y el minuto de un plan, contando solo hasta donde vale. */
export function evaluar(t: Tablas, p: PlanInterno, cam: Camino, ox: number, oy: number): Evaluado & { limite: number } {
  const lim = limite(t, p, cam)
  const a = aproximacion(cam.slice(0, lim), ox, oy, fracciones(t, p).slice(0, lim - 1))
  return { plan: p, distM: a.distM, minuto: a.minuto, limite: lim }
}
```

```ts
/** El primer minuto en que el plan deja de valer, sin tablas: la misma regla que `limite`. */
export function limiteDirecto(c: Campo, aj: AjustesNav, p: PlanInterno, cam: [number, number][], terreno: Terreno | null): number {
  const H = c.minutos
  if (terreno === null) return H + 1
  const z = c.rejilla.altM
  const it = c.rejilla.etiqueta.indexOf('terra')
  const ok = (alt: number, k: number) => !(alt < terreno.techo(cam[k][0], cam[k][1]) + aj.minSobreTerrenoM - 1e-6)
  let k = 0
  for (let i = 0; i < p.franjas.length; i++) {
    const b = p.franjas[i]
    if (i > 0) {
      const a = p.franjas[i - 1]
      const vel = velocidad(aj, z[b] > z[a], p.clases[i - 1])
      const n = minutosTransito(c, a, b, vel)
      for (let j = 1; j <= n; j++) {
        if (++k > H) return H + 1
        const alt = z[a] + Math.sign(z[b] - z[a]) * Math.min(Math.abs(z[b] - z[a]), vel * 60 * j)
        const comprobar = a !== it && b !== it ? true : j === n && b !== it
        if (comprobar && !ok(alt, k)) return k
      }
    }
    const fin = i === p.franjas.length - 1 ? H : k + p.dur[i]
    while (k < fin) {
      k++
      if (b !== it && !ok(z[b], k)) return k
    }
  }
  return H + 1
}
```
